//@version=3
// Trend Signals Toolkit
//
// One indicator, the whole trend kit: a trend-following signal engine with
// two grades, exits, a counter-trend (fade) engine and tri-state candle
// shading; five price overlays (an adaptive trailing stop, a reversion
// ladder, a fast trend stepper, a slow trend keel and a lead / lag trail
// cloud); a take-profit / stop-loss ladder anchored on any of them; and a
// readout panel. A user adds this one script and gets everything; each
// component is a settings section with its own display toggle, and
// everything evaluates on bar close.
//
// Shared primitives, computed once per bar:
//
//   ATR(200): a Wilder average of true range, hardcoded 200 (the engine's
//   atr builtin). Consumers: the trail cloud corridor ((width + 0.5) x),
//   the keel corridors (speed x and 0.5 x that), the TP/SL spacing
//   ((spacing / 2) x, frozen at each anchor).
//
//   The stepper (accelerating mirror-stop stepper on closes): one function
//   with a state slot per instance; slot 1 draws the Trend Stepper overlay
//   at its step-speed setting (slot 0 is spare).
//
//   The signal trigger: ratcheting volatility bands around the bar's average
//   price (ohlc4), 1.9 x ATR(7) x width / 12 wide; a close through the
//   active band flips the signal state. The trend basis (SMA 50 with a
//   0.1 ATR(60) hysteresis), the momentum gauge (Wilder RSI over
//   6 + width / 2 bars) and the 72-bar trend strength feed the signal
//   grades, exits, fade entries, candle shades and the readout panel.
//
// Original implementation. Every feature is covered by a deterministic test harness on two
// engine versions.
//
// Renderer facts (titan-charts-next 0.0.4-alpha.698, measured in the app by
// harness/fill-probe.mjs): a two-value plotLine(fill=true) band fills with its
// FIRST edge color while value1 > value2 and its SECOND edge color while
// value1 < value2, at that edge string's own alpha (an rgba string passes
// through, a plain hex renders at 0.4); the declared fill colors are ignored.
// fillBetween applies its numeric opacity (0..1) and drops the color's alpha.
// So every band below carries its fill alpha in its edge colors
// (opacity(color, pct)) and every fillBetween passes pct / 100.
//
// Presentation: teal / rose signal triangles graded by fill (hollow = normal,
// filled + glow = strong), sky / amber exit crosses, teal / lavender / rose-red
// candles; a green / orange trail with a fading buffer; a violet / sky ladder
// with an outer-denser fill; a lime / coral stepper; an indigo / gold keel
// with a band to price; a cyan / terracotta cloud with a dotted slate lag
// line; mint / salmon TP / SL bands with price tags. Markers ride series plots
// (no drawings); the only drawings are the TP / SL price tags and level lines
// on the last bar.
//
// Series budget: every marker pair shares one plot (absolute placement with
// a per-bar color), every ladder band and TP / SL band is one two-value plot
// with its fill, so the whole toolkit with every option on declares 32
// outputs. Every alert condition is declared unconditionally for the
// components in use (the chart drops the per-bar alert ledger before
// validation).
define(title="Trend Signals Toolkit", position="onchart", axis=false);

// Tape adjustment of a signal in direction sd: +1 when the tape agrees on the

}

// The stepper: an accelerating mirror-stop stepper on closes. Within a leg it
// keeps the extreme close (ep) mirrored across the line; every new extreme
// fires the line toward price with a growing acceleration factor
// (af = 0.5 + step x fires in the leg), and a close crossing the mirrored
// stop flips the leg. One function, one state slot per instance: slot 1 =
// the Trend Stepper overlay (step from its reactivity), slot 0 spare.
// Returns +1 / -1 on a flip, else 0.
persist sDir = [0, 0];
persist sEp = [0, 0];
persist sVal = [0, 0];
persist sAf = [0.5, 0.5];
func sarStep(id, step, price) {
  var flip = 0;
  if (sDir[id] == 0) {
    // Seed: the first bar starts an up leg at the close; the seed washes out
    // at the first one or two flips (the state is one flip event deep).
    sDir[id] = 1;
    sEp[id] = price;
    sVal[id] = price;
    sAf[id] = 0.5;
  } else if (sDir[id] == 1) {
    var stopUp = 2 * sVal[id] - sEp[id];
    if (price < stopUp) {
      sAf[id] = sAf[id] + step;
      sVal[id] = sEp[id] + sAf[id] * (price - sEp[id]);
      sEp[id] = price;
      sDir[id] = -1;
      sAf[id] = 0.5;
      flip = -1;
    } else if (price > sEp[id]) {
      sAf[id] = sAf[id] + step;
      var mirrorUp = 2 * sVal[id] - sEp[id];
      sVal[id] = mirrorUp + sAf[id] * (price - mirrorUp);
      sEp[id] = price;
    }
  } else {
    var stopDn = 2 * sVal[id] - sEp[id];
    if (price > stopDn) {
      sAf[id] = sAf[id] + step;
      sVal[id] = sEp[id] + sAf[id] * (price - sEp[id]);
      sEp[id] = price;
      sDir[id] = 1;
      sAf[id] = 0.5;
      flip = 1;
    } else if (price < sEp[id]) {
      sAf[id] = sAf[id] + step;
      var mirrorDn = 2 * sVal[id] - sEp[id];
      sVal[id] = mirrorDn + sAf[id] * (price - mirrorDn);
      sEp[id] = price;
    }
  }
  return flip;
}

// ADAPTIVE TRAIL
// ============================================================================
// W = mult x ATR(21). Up leg: a new extreme close re-anchors and re-arms the
// stop at full width; otherwise the anchor eases toward price at 1/L per bar
// (the stop backs away on a pullback) and the width tightens toward half at
// the same rate (a stalled trend is protected more closely). trail = anchor
// - width; the buffer edge sits a quarter of the width inside the trail; a
// close through the trail flips the leg and the mirror rule runs.
var trMult = 3.5;
var trRelease = 28;
if (trailPeriod == "1") { trMult = 1.75; trRelease = 14; }
else if (trailPeriod == "2") { trMult = 2.5; trRelease = 20; }
else if (trailPeriod == "4") { trMult = 5; trRelease = 40; }
else if (trailPeriod == "5") { trMult = 7; trRelease = 56; }
var trAlpha = 1 / trRelease;
var trBandFrac = 0.25;
var trFloorFrac = 0.5;
var atr21 = atr(period=21, source=d);

persist trDir = 0;
persist trAnchor = na;
persist trDist = na;
var trFlipUp = false;
var trFlipDn = false;
var trailNow = na;
var trEdgeNow = na;
var trFullWidth = na;
if (needTrail && !isna(atr21)) {
  trFullWidth = trMult * atr21;
  if (trDir == 0) {
    trDir = 1;
    trAnchor = C;
    trDist = trFullWidth;
  } else if (trDir == 1) {
    if (C > trAnchor) {
      trAnchor = C;
      trDist = trFullWidth;
    } else {
      trAnchor = trAnchor + trAlpha * (C - trAnchor);
      trDist = trDist + trAlpha * (trFloorFrac * trFullWidth - trDist);
    }
    if (C < trAnchor - trDist) {
      trDir = -1;
      trAnchor = C;
      trDist = trFullWidth;
      trFlipDn = true;
    }
  } else {
    if (C < trAnchor) {
      trAnchor = C;
      trDist = trFullWidth;
    } else {
      trAnchor = trAnchor + trAlpha * (C - trAnchor);
      trDist = trDist + trAlpha * (trFloorFrac * trFullWidth - trDist);
    }
    if (C > trAnchor + trDist) {
      trDir = 1;
      trAnchor = C;
      trDist = trFullWidth;
      trFlipUp = true;
    }
  }
  if (trDir == 1) {
    trailNow = trAnchor - trDist;
    trEdgeNow = trailNow + trBandFrac * trDist;
  } else {
    trailNow = trAnchor + trDist;
    trEdgeNow = trailNow - trBandFrac * trDist;
  }
}

// ============================================================================
// COMPOSITE VOLUME LADDER (points of control over the last N bars)
// ============================================================================
// A flat grid of volume per price level (the engine has no keyed maps): level
// k of the price grid (bucket low = k x step) sits at grid[k - lvBaseIdx].
// Each bar adds its POINT OF CONTROL (the footprint's busiest bucket, weighted
// by that bucket's volume: three accessor calls, no bucket loop, because a
// loop over every bucket costs 2..3 us per bucket in the interpreter) and
// remembers it in a circular ring of (absolute level, volume) slots, one slot
// per bar; the slot being overwritten is the bar that leaves the window and is
// subtracted first, so the ladder is an exact rolling sum: where the points of
// control clustered over the last N bars. The engine snapshots persisted state
// every bar, so the state stays small and flat: 1201 grid levels (600 each
// side of the current level; the node scan never walks further than 300) and
// two flat ring arrays of N numbers, no nested arrays, no shift. The grid step
// is the first bucket's span (the footprint's own price grid). When price
// walks off the grid it is re-based around the current level; levels that
// fall off the far side are forgotten (their ring slots zeroed).
var lvN = 1201;
var lvHalf = 600;
persist lvStep = na;
persist lvBase = na;
persist lvBaseIdx = 0;
persist lvGrid = [];
persist lvRingK = [];
persist lvRingV = [];
persist lvRingPos = 0;
persist lvBars = 0;
if (needVp) {
  timeseries vp = source("volume_profile", symbol=currentSymbol, exchange=currentExchange, ticksPerBar=ofNodeTicks, currency="USD");
  var lvCount = vpBucketCount(vp);
  var lvK = 0;
  var lvVol = 0;
  if (isnum(lvCount) && lvCount > 0) {
    if (isna(lvStep)) {
      var lvFirst = vp[0][1];
      if (lvFirst[1] - lvFirst[0] > 0) {
        lvStep = lvFirst[1] - lvFirst[0];
        lvBaseIdx = math.round(lvFirst[0] / lvStep) - lvHalf;
        lvBase = lvBaseIdx * lvStep;
        for (var g0 = 0; g0 < lvN; g0 = g0 + 1) { lvGrid.push(0); }
        for (var r0 = 0; r0 < ofNodeWindow; r0 = r0 + 1) { lvRingK.push(0); lvRingV.push(0); }
      }
    }
    var lvPoc = vpPoc(vp);
    var lvPocVol = vpPocVolume(vp);
    if (!isna(lvStep) && isnum(lvPoc) && isnum(lvPocVol) && lvPocVol > 0) {
      // the point of control is a bucket center: its level = round((poc - step / 2) / step)
      lvK = math.round((lvPoc - 0.5 * lvStep) / lvStep);
      lvVol = lvPocVol;
    }
  }
  if (!isna(lvStep)) {
    // the slot being overwritten is the bar leaving the window
    var lvOldK = lvRingK[lvRingPos];
    var lvOldV = lvRingV[lvRingPos];
    if (lvOldV > 0) {
      var lvOldIdx = lvOldK - lvBaseIdx;
      if (lvOldIdx >= 0 && lvOldIdx < lvN) { lvGrid[lvOldIdx] = lvGrid[lvOldIdx] - lvOldV; }
      lvBars = lvBars - 1;
    }
    if (lvVol > 0) {
      var lvIdx = lvK - lvBaseIdx;
      if (lvIdx < 0 || lvIdx >= lvN) {
        var lvShift = lvIdx - lvHalf;
        var lvNew = [];
        for (var g1 = 0; g1 < lvN; g1 = g1 + 1) {
          var gSrc = g1 + lvShift;
          lvNew.push(gSrc >= 0 && gSrc < lvN ? lvGrid[gSrc] : 0);
        }
        lvGrid = lvNew;
        lvBaseIdx = lvBaseIdx + lvShift;
        lvBase = lvBaseIdx * lvStep;
        for (var ri = 0; ri < ofNodeWindow; ri = ri + 1) {
          if (lvRingV[ri] > 0) {
            var rIdx = lvRingK[ri] - lvBaseIdx;
            if (rIdx < 0 || rIdx >= lvN) { lvRingV[ri] = 0; lvBars = lvBars - 1; }
          }
        }
        lvIdx = lvHalf;
      }
      lvGrid[lvIdx] = lvGrid[lvIdx] + lvVol;
      lvBars = lvBars + 1;
    }
    lvRingK[lvRingPos] = lvK;
    lvRingV[lvRingPos] = lvVol;
    lvRingPos = lvRingPos + 1;
    if (lvRingPos >= ofNodeWindow) { lvRingPos = 0; }
  }
}
// Node scan: from fromPrice outward (sideDir +1 up / -1 down) to toPrice, a
// level is a candidate node when it holds more volume than its lower neighbour
// and at least as much as its upper one (peaks 2 steps apart or closer keep
// the higher); nodes are the candidates holding at least 20% of the largest
// level in the span, taken outward. Writes up to 3 node prices (bucket
// centers) into lvFound and returns how many.
persist lvFound = [0, 0, 0];
func scanNodes(sideDir, fromPrice, toPrice) {
  var found = 0;
  var iFrom = math.round((fromPrice - lvBase) / lvStep);
  var iTo = math.round((toPrice - lvBase) / lvStep);
  var steps = sideDir * (iTo - iFrom) + 1;
  if (steps > 300) { steps = 300; }
  var cand = [];
  var candVol = [];
  var lastIdx = -10;
  var spanMax = 0;
  for (var st = 0; st < steps; st = st + 1) {
    var gi = iFrom + sideDir * st;
    if (gi >= 1 && gi <= lvN - 2) {
      var gv = lvGrid[gi];
      if (gv > spanMax) { spanMax = gv; }
      if (gv > 0 && gv > lvGrid[gi - 1] && gv >= lvGrid[gi + 1]) {
        if (cand.length > 0 && math.abs(gi - lastIdx) <= 2) {
          if (gv > candVol[cand.length - 1]) { cand[cand.length - 1] = gi; candVol[cand.length - 1] = gv; lastIdx = gi; }
        } else if (cand.length < 12) {
          cand.push(gi);
          candVol.push(gv);
          lastIdx = gi;
        }
      }
    }
  }
  var floorVol = 0.2 * spanMax;
  for (var ci = 0; ci < cand.length; ci = ci + 1) {
    if (found < 3 && candVol[ci] >= floorVol) {
      lvFound[found] = lvBase + (cand[ci] + 0.5) * lvStep;
      found = found + 1;
    }
  }
  return found;
}
persist lvTick = 8;
persist lvUpN = 0;
persist lvDnN = 0;
persist lvUp = [0, 0, 0];
persist lvDn = [0, 0, 0];

      }

        lzUp1Now = lvUp[0];

        lzUp3Now = lvUpN > 2 ? lvUp[2] : lzUp2Now + 0.5 * lzOffset;
      }

      }
    }

    else if (C < lzDn3Now) { lzZone = -3; }
    else if (C < lzDn2Now) { lzZone = -2; }
    else if (C < lzDn1Now) { lzZone = -1; }
    if (C > lzBalanceNow) { lzSide = 1; }
    else if (C < lzBalanceNow) { lzSide = -1; }
  }
}

lzPrevSide = lzSide;
timeseries ladderBalanceLine = lzBalanceNow;

// ============================================================================
// TREND KEEL
// ============================================================================
// Stage 1, the tracker, trails the close inside a corridor of
// reactivity x ATR(200). Stage 2, the keel, trails the CURRENT bar's tracker
// inside a 0.5x corridor and only wakes once the tracker's accumulated
// excursion beyond that corridor (summed over same-direction push bars in
// ATR(200) units, kept through quiet bars, reset when the push direction
// changes) reaches 1 ATR; while awake it follows every push, a non-push bar
// puts it back to sleep. State = sign of the keel's last step.
var klMult = 1.75;
if (keelReactivity == "1") { klMult = 0.5; }
else if (keelReactivity == "2") { klMult = 1.0; }
else if (keelReactivity == "4") { klMult = 2.5; }
else if (keelReactivity == "5") { klMult = 3.5; }
var klRatio = 0.5;
var klWakeDebt = 1.0;
persist klTracker = na;
persist klKeel = na;
persist klDir = 0;
persist klDebt = 0;
persist klPushDir = 0;
persist klAwake = false;
var klStepUp = false;
var klStepDn = false;
var klFlipUp = false;
var klFlipDn = false;
var keelNow = na;
if (needKeel && atrReady) {
  var klTrackerWidth = klMult * atr200;
  var klKeelWidth = klRatio * klMult * atr200;
  if (isna(klTracker)) {
    klTracker = C;
    klKeel = C;
  } else {
    if (C - klTrackerWidth > klTracker) {
      klTracker = C - klTrackerWidth;
    } else if (C + klTrackerWidth < klTracker) {
      klTracker = C + klTrackerWidth;
    }
    var klPd = 0;
    if (klTracker - klKeelWidth > klKeel) {
      klPd = 1;
    } else if (klTracker + klKeelWidth < klKeel) {
      klPd = -1;
    }
    if (klPd == 0) {
      klAwake = false;
    } else {
      if (klPd != klPushDir) {
        klDebt = 0;
        klPushDir = klPd;
      }
      klDebt = klDebt + (math.abs(klTracker - klKeel) / atr200 - klRatio * klMult);
      if (klAwake || klDebt >= klWakeDebt) {
        klKeel = klPd == 1 ? klTracker - klKeelWidth : klTracker + klKeelWidth;
        klAwake = true;
        klDebt = 0;
        if (klPd == 1) {
          klStepUp = true;
          if (klDir == -1) { klFlipUp = true; }
        } else {
          klStepDn = true;
          if (klDir == 1) { klFlipDn = true; }
        }
        klDir = klPd;
      }
    }
  }
  keelNow = klKeel;
}
timeseries keel = keelNow;
var klIsBull = klDir >= 0;

// ============================================================================
// TRAIL CLOUD
// ============================================================================
// Corridor width (period + 0.5) x ATR(200). The lead trails the close (hold
// unless dragged to a corridor edge); the lag trails the CURRENT bar's lead
// with the same width. Cloud state: bullish while lead > lag.
var ncMult = tonumber(cloudPeriod) + 0.5;
persist ncLead = 0;
persist ncLag = 0;
persist ncReady = false;
persist ncState = 0;
var ncLeadNow = na;
var ncLagNow = na;
var ncLeadStepUp = false;
var ncLeadStepDown = false;
var ncLagStepUp = false;
var ncLagStepDown = false;
var ncChanged = false;
var ncFlippedBull = false;
var ncFlippedBear = false;
if (needCloud && atrReady) {
  var ncDist = ncMult * atr200;
  if (!ncReady) {
    ncLeadNow = C;
    ncLagNow = C;
    ncReady = true;
  } else {
    ncLeadNow = ncLead;
    if (C - ncDist > ncLead) {
      ncLeadNow = C - ncDist;
      ncLeadStepUp = true;
    } else if (C + ncDist < ncLead) {
      ncLeadNow = C + ncDist;
      ncLeadStepDown = true;
    }
    ncLagNow = ncLag;
    if (ncLeadNow - ncDist > ncLag) {
      ncLagNow = ncLeadNow - ncDist;
      ncLagStepUp = true;
    } else if (ncLeadNow + ncDist < ncLag) {
      ncLagNow = ncLeadNow + ncDist;
      ncLagStepDown = true;
    }
  }
  ncLead = ncLeadNow;
  ncLag = ncLagNow;
  var ncNow = ncState;
  if (ncLeadNow > ncLagNow) {
    ncNow = 1;
  } else if (ncLeadNow < ncLagNow) {
    ncNow = -1;
  }
  ncChanged = ncNow != ncState;
  ncFlippedBull = ncNow == 1 && ncState == -1;
  ncFlippedBear = ncNow == -1 && ncState == 1;
  ncState = ncNow;
}
timeseries cloudLead = ncLeadNow;
timeseries cloudLag = ncLagNow;
var ncIsBull = ncState >= 0;
// Price position vs the lead line (the cloud filter's rule), held through touches.
persist ncPricePos = 0;
if (!isna(ncLeadNow)) {
  if (C > ncLeadNow) { ncPricePos = 1; }
  else if (C < ncLeadNow) { ncPricePos = -1; }
}

// ============================================================================
// TREND SIGNALS
// ============================================================================
// Model constants (not knobs): trigger band 1.9 x ATR(7) at sensitivity 12
// (scales linearly with sensitivity), basis SMA(50) with a 0.1 ATR(60)
// hysteresis, stretch unit ATR(60), trend strength window 72, gauge
// thresholds 35/65 (contrarian), 25/75 (contrarian +), 70/30 (exit climax),
// 55/45 (exit re-arm), 50 (contrarian re-arm and exit).
var sgX = 1.9 * sensitivity / 12;
var sgHystK = 0.1;
var gLenRaw = math.round(6 + sensitivity / 2);
var gLen = gLenRaw < 3 ? 3 : (gLenRaw > 100 ? 100 : gLenRaw);

timeseries basis = sma(source=d.close, period=50);
timeseries stretchUnit = atr(period=60, source=d);
timeseries sgMid = (d.open + d.high + d.low + d.close) / 4;
var sgAtr = atr(period=7, source=d);
timeseries upMove = d.close > d.close[1] ? d.close - d.close[1] : 0;
timeseries dnMove = d.close < d.close[1] ? d.close[1] - d.close : 0;
timeseries avgUp = rma(source=upMove, period=gLen);
timeseries avgDn = rma(source=dnMove, period=gLen);
timeseries gauge = avgDn == 0 ? 100 : (avgUp == 0 ? 0 : 100 - 100 / (1 + avgUp / avgDn));
timeseries one = 1;
timeseries idx = cum(source=one);
timeseries rr = correlation(s1=d.close, s2=idx, period=72);
timeseries strength = 100 * rr * rr;
timeseries strengthRegimeLine = 50;

// 1. The trigger (signal state): ratcheting volatility bands around the bar's
// average price. The lower band ratchets up while the previous close held
// above it (the upper band ratchets down while the close held below it); a
// close through the active band flips the state. Every flip is a signal.
persist sgDir = 0;
persist sgLower = na;
persist sgUpper = na;
var flipUp = false;
var flipDn = false;
if (!isna(sgAtr)) {
  var sgLo = sgMid - sgX * sgAtr;
  var sgHi = sgMid + sgX * sgAtr;
  var sgPrevClose = d.close[1];
  if (!isna(sgLower) && !isna(sgPrevClose) && sgPrevClose > sgLower && sgLo < sgLower) { sgLo = sgLower; }
  if (!isna(sgUpper) && !isna(sgPrevClose) && sgPrevClose < sgUpper && sgHi > sgUpper) { sgHi = sgUpper; }
  if (sgDir == 0) { sgDir = 1; }
  else if (sgDir == 1 && !isna(sgLower) && C < sgLower) { sgDir = -1; flipDn = true; }
  else if (sgDir == -1 && !isna(sgUpper) && C > sgUpper) { sgDir = 1; flipUp = true; }
  sgLower = sgLo;
  sgUpper = sgHi;
}
var dir = sgDir;

// 2. The basis: side of the 50-bar average with a 0.1 ATR(60) hysteresis
// band, held while the close sits inside the band.
persist side = 0;
var basisNow = basis;
var stretchUnitNow = stretchUnit;
if (!isna(basisNow)) {
  var sgBand = isna(stretchUnitNow) ? 0 : sgHystK * stretchUnitNow;
  if (C > basisNow + sgBand) { side = 1; }
  else if (C < basisNow - sgBand) { side = -1; }
}
var strengthNow = strength;
var tsOk = !filterTs || (!isna(strengthNow) && strengthNow > 50);

// Overlay filters: a signal in direction sd is kept when the overlay agrees.
func overlayAgrees(sd) {
  var ok = true;
  if (filterTrail && trDir != sd) { ok = false; }
  if (filterStepper && stDir != sd) { ok = false; }
  if (filterKeel && klDir != sd) { ok = false; }
  if (filterCloud && ncPricePos != sd) { ok = false; }
  return ok;
}

// Trend signals: every flip, graded by the basis; normal signals can
// turn strong later. Filters suppress the marker (and the position it opens).
persist openDir = 0;
persist entry = na;
persist graded = false;
persist exitArmedB = true;
persist exitArmedS = true;
// Tape grade of the open position: sigAdj (-1 / 0 / +1) from tapeAdjust on
// the signal bar, sigTag from open interest; the marker shade of a signal
// fired this bar rides sigShade (3 = full color when the tape is silent).
persist sigAdj = 0;
persist sigTag = "";
var sigShade = 3;
var sigBull = false;
var sigBear = false;
var sigBullP = false;
var sigBearP = false;
var turnBull = false;
var turnBear = false;
if (flipUp || flipDn) {
  var sd = flipUp ? 1 : -1;
  var plus = side == sd;
  var shown = tsOk && (!filterBasis || plus) && overlayAgrees(sd);
  if (shown) {
    if (sd == 1) {
      if (plus) { sigBullP = true; } else { sigBull = true; }
    } else {
      if (plus) { sigBearP = true; } else { sigBear = true; }
    }
    openDir = sd;
    entry = C;
    graded = plus;
    sigAdj = tapeAdjust(sd);
    sigTag = flowTag(sd);
    if (tapeReady) { sigShade = clampGrade((plus ? 3 : 2) + sigAdj) - 1; }
  } else {
    openDir = 0;
    entry = na;
    graded = true;
    sigAdj = 0;
    sigTag = "";
  }
  exitArmedB = true;
  exitArmedS = true;
} else if (openDir != 0 && !graded && side == openDir) {
  if (openDir == 1) { turnBull = true; } else { turnBear = true; }
  graded = true;
}
var bcv = (side != 0 && side == dir) ? dir : 0;

// 3. The momentum gauge: confirmation exits, contrarian signals and exits.
persist cArmB = true;
persist cArmS = true;
persist cArmBP = true;
persist cArmSP = true;
persist cOpenDir = 0;
persist cEntry = na;
persist cExitDone = true;
persist cPlus = false;
persist cAdj = 0;
persist cTag = "";
var cShade = 3;
var g = gauge;
var gp = gauge[1];
var bullExit = false;
var bearExit = false;
var cBull = false;
var cBullP = false;
var cBear = false;
var cBearP = false;
if (!isna(g)) {
  if (g < 55) { exitArmedB = true; }
  if (g > 45) { exitArmedS = true; }
  if (!isna(gp) && confirmMode && openDir == dir) {
    if (openDir == 1 && exitArmedB && g > 70 && gp <= 70 && C > entry) { bullExit = true; exitArmedB = false; }
    if (openDir == -1 && exitArmedS && g < 30 && gp >= 30 && C < entry) { bearExit = true; exitArmedS = false; }
  }
  if (g > 50) { cArmB = true; cArmBP = true; }
  if (g < 50) { cArmS = true; cArmSP = true; }
  var prevClose = d.close[1];
  if (!isna(prevClose)) {
    var counterDown = C < prevClose;
    var counterUp = C > prevClose;
    var fired = 0;
    var firedPlus = false;
    if (counterDown && g < 35 && (cArmB || (cArmBP && g < 25))) {
      fired = 1;
      firedPlus = g < 25;
    } else if (counterUp && g > 65 && (cArmS || (cArmSP && g > 75))) {
      fired = -1;
      firedPlus = g > 75;
    }
    if (fired != 0) {
      if (fired == 1) {
        cArmB = false;
        if (firedPlus) { cArmBP = false; }
      } else {
        cArmS = false;
        if (firedPlus) { cArmSP = false; }
      }
      var cShown = tsOk && (!filterBasis || side == fired) && overlayAgrees(fired);
      if (cShown) {
        if (fired == 1) {
          if (firedPlus) { cBullP = true; } else { cBull = true; }
        } else {
          if (firedPlus) { cBearP = true; } else { cBear = true; }
        }
        cOpenDir = fired;
        cEntry = C;
        cExitDone = false;
        cPlus = firedPlus;
        cAdj = tapeAdjust(fired);
        cTag = flowTag(fired);
        if (tapeReady) { cShade = clampGrade((firedPlus ? 3 : 2) + cAdj) - 1; }
      }
    }
  }
  if (!isna(gp) && contraMode && !cExitDone) {
    if (cOpenDir == 1 && g > 50 && gp <= 50 && C > cEntry) { bullExit = true; cExitDone = true; }
    if (cOpenDir == -1 && g < 50 && gp >= 50 && C < cEntry) { bearExit = true; cExitDone = true; }
  }
}
// The flip bar closes the leg for both engines (a fade firing on the flip bar
// was graded on the full leg above); the next leg starts on the following bar.
if (flipUp || flipDn) {
  legDelta = 0;
  legTotal = 0;
}
// Grade of the open position on the 1..4 scale: base 2 (normal) or 3
// (strong, including a later upgrade) plus the tape adjustment of its signal
// bar; the tag rides along. Both feed the readout and the alert messages.
var posGrade = confirmMode ? clampGrade((graded ? 3 : 2) + sigAdj) : clampGrade((cPlus ? 3 : 2) + cAdj);
var posTag = confirmMode ? sigTag : cTag;
// Built on signal event bars only: the engine evaluates every alert message
// on every bar, and a fired alert only ever reads the note of its own bar.
var sigEventBar = sigBull || sigBullP || sigBear || sigBearP || turnBull || turnBear || cBull || cBullP || cBear || cBearP;
var gradeNote = (ofGradeSignals && sigEventBar) ? ", grade " + tostring(posGrade) + "/4" + (posTag != "" ? ", " + posTag : "") : "";
// The signal events the TP/SL ladder can anchor on (the active mode's signals).
var sigEvent = 0;
if (confirmMode) {
  if (sigBull || sigBullP) { sigEvent = 1; }
  else if (sigBear || sigBearP) { sigEvent = -1; }
} else if (contraMode) {
  if (cBull || cBullP) { sigEvent = 1; }
  else if (cBear || cBearP) { sigEvent = -1; }
}

// ============================================================================
// TP/SL LADDER
// ============================================================================
// On a trigger event (direction e): dir = e, M = close, W = (Distance/2) x
// ATR(200) frozen. Else if the close runs strictly beyond the outermost level
// (M + dir x 2W with the trend, M - dir x 2W against it): M = close, W
// refrozen, dir kept. Levels TP1 = M + dir W, TP2 = M + dir 2W, SL1 = M - dir
// W, SL2 = M - dir 2W; the next uncrossed level per side is re-evaluated on
// close (TP1 until the close is beyond it, then TP2; same on the stop side).
var tpK = tpslDistance / 2;
var tpEvt = 0;
if (needTpsl) {
  if (tpslSrc == "Signals") { tpEvt = sigEvent; }
  else if (tpslSrc == "Adaptive Trail") { tpEvt = trFlipUp ? 1 : (trFlipDn ? -1 : 0); }
  else if (tpslSrc == "Reversion Ladder") { tpEvt = lzReachedUpper3 ? 1 : (lzReachedLower3 ? -1 : 0); }
  else if (tpslSrc == "Trend Stepper") { tpEvt = stFlip; }
  else if (tpslSrc == "Trend Keel") { tpEvt = klFlipUp ? 1 : (klFlipDn ? -1 : 0); }
  else if (tpslSrc == "Trail Cloud") { tpEvt = ncChanged ? ncState : 0; }
}
persist tpDir = 0;
persist tpM = 0;
persist tpHalfW = 0;
var tpNewAnchor = 0;
if (needTpsl) {
  if (tpEvt != 0 && atrReady) {
    tpDir = tpEvt;
    tpM = C;
    tpHalfW = tpK * atr200;
    tpNewAnchor = 1;
  } else if (tpDir != 0) {
    var tpOuterPrev = tpM + tpDir * 2 * tpHalfW;
    var slOuterPrev = tpM - tpDir * 2 * tpHalfW;
    if (tpDir * (C - tpOuterPrev) > 0 || tpDir * (slOuterPrev - C) > 0) {
      tpM = C;
      tpHalfW = tpK * atr200;
      tpNewAnchor = 2;
    }
  }
}
var tpLive = needTpsl && tpDir != 0;
timeseries tpInner = tpLive ? tpM + tpDir * tpHalfW : na;
timeseries tpOuter = tpLive ? tpM + tpDir * 2 * tpHalfW : na;
timeseries slInner = tpLive ? tpM - tpDir * tpHalfW : na;
timeseries slOuter = tpLive ? tpM - tpDir * 2 * tpHalfW : na;
// Progress flags: 1 once the close is beyond the inner level of that side.
var tpProgressed = tpLive ? (tpDir * (C - (tpM + tpDir * tpHalfW)) > 0 ? 1 : 0) : 0;
var slProgressed = tpLive ? (tpDir * ((tpM - tpDir * tpHalfW) - C) > 0 ? 1 : 0) : 0;
timeseries activeTP = tpLive ? (tpProgressed == 1 ? tpM + tpDir * 2 * tpHalfW : tpM + tpDir * tpHalfW) : na;
timeseries activeSL = tpLive ? (slProgressed == 1 ? tpM - tpDir * 2 * tpHalfW : tpM - tpDir * tpHalfW) : na;
persist tpProgPrev = 0;
persist slProgPrev = 0;
var tp1Reached = tpNewAnchor == 0 && tpProgressed == 1 && tpProgPrev == 0;
var sl1Breached = tpNewAnchor == 0 && slProgressed == 1 && slProgPrev == 0;
tpProgPrev = tpProgressed;
slProgPrev = slProgressed;

// ============================================================================
// VIEW
// ============================================================================
// Fill opacities ride the 0..1 scale the chart renderer reads.

// ---- candle coloring ----
var stretch = (!isna(basisNow) && !isna(stretchUnitNow) && stretchUnitNow > 0) ? (C - basisNow) / stretchUnitNow : na;
var candleColor = neutralCandle;
if (coloring == "Confirmation Simple") {
  candleColor = bcv == 1 ? strongBullCandle : (bcv == -1 ? strongBearCandle : neutralCandle);
} else if (coloring == "Confirmation Gradient") {
  if (!isna(stretch)) {
    var raw = dir * stretch / 2;
    var intensity = raw < 0 ? 0 : (raw > 1 ? 1 : raw);
    candleColor = blend(neutralCandle, dir == 1 ? strongBullCandle : strongBearCandle, intensity);
  }
} else if (coloring == "Contrarian Gradient") {
  if (!isna(g)) {
    var osRaw = (50 - g) / 25;
    var os = osRaw < -1 ? -1 : (osRaw > 1 ? 1 : osRaw);
    candleColor = os > 0 ? blend(neutralCandle, strongBullCandle, os) : blend(neutralCandle, strongBearCandle, 0 - os);
  }
} else if (deltaColoring) {
  // Real aggression: the 3-bar taker delta share in the signal's direction,
  // full color at 30 points; the inferred gradient where the tape is silent.
  if (sidedNow && isnum(share3Now)) {
    var dRaw = dir * share3Now / 0.3;
    var dInt = dRaw < 0 ? 0 : (dRaw > 1 ? 1 : dRaw);
    candleColor = blend(neutralCandle, dir == 1 ? strongBullCandle : strongBearCandle, dInt);
  } else if (!isna(stretch)) {
    var dRaw2 = dir * stretch / 2;
    var dInt2 = dRaw2 < 0 ? 0 : (dRaw2 > 1 ? 1 : dRaw2);
    candleColor = blend(neutralCandle, dir == 1 ? strongBullCandle : strongBearCandle, dInt2);
  }
}
if (coloring != "None") {
  barcolor(color=candleColor);
}

// ---- overlays (drawn under the markers) ----
if (trailOn) {
  var trStateColor = trIsBull ? trailBullColor : trailBearColor;
  // Tightness 0 = stop at full width (fresh extreme), 1 = fully tightened.
  var trTightness = 0;
  if (!isna(trDist) && !isna(trFullWidth) && trFullWidth > 0) {
    trTightness = (trFullWidth - trDist) / (trFullWidth * (1 - trFloorFrac));
    if (trTightness < 0) { trTightness = 0; }
    if (trTightness > 1) { trTightness = 1; }
  }
  // The renderer joins fillBetween rows into one polygon only while consecutive
  // bars share the color, and drops single-bar runs, so the fade is quantized to
  // three steps (a per-bar blend would leave nothing to draw).
  var trFadeStep = trTightness < 0.33 ? 0 : (trTightness < 0.66 ? 0.35 : 0.7);
  var trBufferColor = trailFade ? blend(trStateColor, "#8a93a6", trFadeStep) : trStateColor;
  if (trailBuffer) {
    fillBetween(series1=trail, series2=trailEdgeLine, color=trBufferColor, opacity=trailBufferOpacity / 100);
  }
  plotLine(value=trail, colors=[trailBullColor, trailBearColor], colorIndex=trIsBull ? 0 : 1,
    width=2, smooth=false,
    label=["Adaptive Trail"],
    desc=["Volatility-responsive trailing stop: one width (multiplier x ATR(21)) from the leg's extreme close, easing away from price on pullbacks and tightening while the trend stalls. Flips when the close crosses it."]);
  if (trailEdge) {
    plotLine(value=trailEdgeLine, colors=[trailBullColor, trailBearColor], colorIndex=trIsBull ? 0 : 1,
      width=1, smooth=false, lineStyle="dotted", showPriceDisplay=false,
      label=["Trail Buffer Edge"],
      desc=["Far edge of the buffer: a quarter of the current width inside the trail. A dip into the buffer that closes back on the trend side is a test, not a flip."]);
  }

if (ladderOn) {
  // Band fill = edge color at the edge's alpha (renderer fact above): the
  // rung edges carry the zone shade, the outer band 1.75x denser.
  var lzInnerFill = opacity(ladderUpperColor, ladderOpacity);
  var lzOuterPct = ladderOpacity * 1.75 > 100 ? 100 : ladderOpacity * 1.75;
  var lzInnerFillDn = opacity(ladderLowerColor, ladderOpacity);
  var lzOuterFill = opacity(ladderUpperColor, lzOuterPct);
  var lzOuterFillDn = opacity(ladderLowerColor, lzOuterPct);
  plotLine(value=[ladderUp1, ladderUp2], fill=true, colors=[lzInnerFill, lzInnerFill, lzInnerFill, lzInnerFill],
    width=1, smooth=false, showPriceDisplay=false,
    label=["Upper Rung 1", "Upper Rung 2"],
    desc=["Inner edge of the top zone: balance + (2 + 0.6 x spacing) x ATR(14 x spacing). A close above it is the first over-extension warning.", "Middle rung of the top zone at 1.5x the inner offset."]);
  plotLine(value=[ladderUp2, ladderUp3], fill=true, colors=[lzOuterFill, lzOuterFill, lzOuterFill, lzOuterFill],
    width=1, smooth=false, showPriceDisplay=false,
    label=["Upper Rung 2 (outer band)", "Upper Rung 3"],
    desc=["Middle rung of the top zone (the outer band starts here).", "Outer rung of the top zone at 2x the inner offset: the most stretched level the ladder marks. Price can still run past it in a strong trend."]);
  plotLine(value=[ladderDn2, ladderDn1], fill=true, colors=[lzInnerFillDn, lzInnerFillDn, lzInnerFillDn, lzInnerFillDn],
    width=1, smooth=false, showPriceDisplay=false,
    label=["Lower Rung 2", "Lower Rung 1"],
    desc=["Middle rung of the bottom zone at 1.5x the inner offset.", "Inner edge of the bottom zone: balance - (2 + 0.6 x spacing) x ATR(14 x spacing). A close below it is the first over-extension warning."]);
  plotLine(value=[ladderDn3, ladderDn2], fill=true, colors=[lzOuterFillDn, lzOuterFillDn, lzOuterFillDn, lzOuterFillDn],
    width=1, smooth=false, showPriceDisplay=false,
    label=["Lower Rung 3", "Lower Rung 2 (outer band)"],
    desc=["Outer rung of the bottom zone at 2x the inner offset: the most stretched level the ladder marks. Price can still run past it in a strong sell-off.", "Middle rung of the bottom zone (the outer band starts here)."]);
  if (ladderBalance) {
    plotLine(value=ladderBalanceLine, colors=[ladderBalanceColor], width=1, smooth=false, lineStyle="dotted",
      label=["Balance Line"],

if (stepperOn) {
  plotLine(value=stepperLine, colors=[stepperBullColor, stepperBearColor], colorIndex=stIsBull ? 0 : 1,

      rows.push([{value: "TP/SL"}, {value: tpText, textColor: tpDir == 1 ? tpColor : (tpDir == -1 ? slColor : anchorColor), align: "right", tooltip: "Direction of the active ladder and its next uncrossed levels."}]);
    }
  }
  plotTable(data=rows, position=pos, headerRow=true, fontSize=fontSize,
    backgroundColor=dashColor, textColor="#c2cad6", borderColor="#2c3440", borderWidth=1);
}
